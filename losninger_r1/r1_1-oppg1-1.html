\begin{align} a) &{} \quad 2^3\cdot2^2 = 2^{3+2} = 2^5 = 32 \\[1em] b) &{} \quad 2\cdot 2^3 = 2^{1+3} = 2^4 = 16 \\[1em] c) &{} \quad \left(2^{-1}\cdot4\right)^2 = \left(\frac{4}{2^1}\right)^2 = 2^2 = 4 \\[1em] d) &{} \quad \left(2^2 \cdot 2\right)^2 + 2^3 = \left(2^3\right)^2 + 2^3 = 2^{3\cdot2} + 2^3 \\[0.5em] &{} \qquad = 2^6+2^3 = 64+8=72 \\[1em] e) &{} \quad \left(2^2\cdot2\right)^2\cdot3 = \left(4+2\right)^2\cdot3=6^2\cdot3=36\cdot3=108 \\[1em] f) &{} \quad \frac{3^4}{3^2} = 3^{4-2} = 3^2 = 9 \\[1em] g) &{} \quad \frac{2^3 \cdot 3^4}{3^3 \cdot2} = 3^{4-3} \cdot 2^{3-1} = 3\cdot2^2 = 12 \\[1em] h) &{} \quad 2^{-1} \cdot \left( \frac{3^2}{3^3}\right)^2 = \frac{1}{2} \cdot \left(3^{2-3}\right)^2 = \frac{1}{2} \cdot \left(3^{-1}\right)^2\\[0.5em] &{} \qquad = \frac{1}{2} \cdot 3^{-2} = \frac{1}{2} \cdot \frac{1}{3^2} = \frac{1}{18} \end{align}
